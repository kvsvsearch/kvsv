rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /stories/{storyId} {
      allow create: if request.auth != null && request.auth.uid == 'RESEARCHER_UID';
      allow update: if request.auth != null && request.auth.uid == 'RESEARCHER_UID';
      allow delete: if request.auth != null && request.auth.uid == 'RESEARCHER_UID';
      allow read: if request.auth != null && request.auth.uid == 'RESEARCHER_UID'
                  || resource.data.approved_public == true;
    }
  }
}
